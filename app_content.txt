import React, { useState, useEffect, useRef, useCallback } from 'react';
import './App.css';

const App = () => {
  // ===================== COLOR SCHEME =====================
  const C = {
    bg: "#03080c",
    bgPanel: "#0a0f14",
    bgInput: "#0d1218",
    border: "#1a2332",
    text: "#e4e6eb",
    textSecondary: "#9ca3af",
    accent: "#e8a020",
    accentLight: "#f0b23b",
    green: "#00c853",
    red: "#ff1744",
    yellow: "#ffd600",
    blue: "#1e88e5",
    purple: "#9c27b0",
    hover: "#1a2332",
  };

  // ===================== DROPDOWNS DATA =====================
  const PAIRS = [
    "EUR/USD", "GBP/USD", "USD/JPY", "AUD/USD", "USD/CAD",
    "EUR/GBP", "EUR/JPY", "GBP/JPY", "CHF/USD", "NZD/USD",
    "EUR/CHF", "AUD/JPY", "GBP/AUD", "XAU/USD", "BTC/USD"
  ];

  const SESSIONS = [
    "Asian Session", "London Session", "New York Session", "Sydney Session", "Overnight"
  ];

  const SETUPS = [
    "Support/Resistance", "Trend Following", "Breakout", "Range Trading",
    "Pullback", "Scalp", "News Trade", "Other"
  ];

  const EMOTIONS = [
    "Confident", "Neutral", "Anxious", "Frustrated",
    "Greedy", "Fearful", "Impatient", "Focused"
  ];

  const GRADES = ["A+", "A", "B", "C", "D", "F"];
  const TIMEFRAMES = ["M1", "M5", "M15", "M30", "H1", "H4", "D1", "W1"];

  // ===================== STATE MANAGEMENT =====================
  const [trades, setTrades] = useState([]);
  const [dataLoaded, setDataLoaded] = useState(false);
  const [currentPage, setCurrentPage] = useState('dashboard');
  const [showSaveIndicator, setShowSaveIndicator] = useState(false);
  const [saveStatus, setSaveStatus] = useState('saved'); // 'saved' | 'saving' | 'error'
  const saveTimeoutRef = useRef(null);
  const autoSaveTimeoutRef = useRef(null);

  // Form state for new trade
  const emptyForm = {
    date: new Date().toISOString().split('T')[0],
    pair: PAIRS[0],
    direction: 'Long',
    entryTime: '09:00',
    entryPrice: '',
    exitTime: '10:00',
    exitPrice: '',
    stopLoss: '',
    takeProfit: '',
    riskAmount: '',
    rewardAmount: '',
    lotSize: '',
    pipsGained: '',
    session: SESSIONS[0],
    setup: SETUPS[0],
    confluences: [],
    emotion: EMOTIONS[0],
    mistake: '',
    outcome: 'Win',
    tradeIdea: '',
    executionNotes: '',
    lessonLearned: '',
    grade: GRADES[0],
    timeframe: TIMEFRAMES[4],
  };

  const [formData, setFormData] = useState(emptyForm);
  const [editingId, setEditingId] = useState(null);
  const [searchTerm, setSearchTerm] = useState('');
  const [filterPair, setFilterPair] = useState('All Pairs');
  const [summaryPeriod, setSummaryPeriod] = useState('week'); // 'week' | 'month' | 'all'

  // ===================== MT5 STATE =====================
  const [mt5Connected, setMt5Connected] = useState(false);
  const [mt5Account, setMt5Account] = useState(null);
  const [mt5OpenTrades, setMt5OpenTrades] = useState([]);
  const [mt5History, setMt5History] = useState([]);
  const [mt5Summary, setMt5Summary] = useState(null);
  const [mt5Loading, setMt5Loading] = useState(false);
  const [mt5Error, setMt5Error] = useState('');
  const [mt5HistoryDays, setMt5HistoryDays] = useState(30);
  const [mt5ActiveAccount, setMt5ActiveAccount] = useState(0);
  const [mt5Switching, setMt5Switching] = useState(false);
  const [mt5Accounts, setMt5Accounts] = useState([]);
  const mt5RefreshRef = useRef(null);

  const MT5_URL = 'http://127.0.0.1:5000';

  const fetchMt5Data = useCallback(async () => {
    try {
      const statusRes = await fetch(`${MT5_URL}/status`);
      const status = await statusRes.json();
      if (!status.connected) {
        setMt5Connected(false);
        setMt5Error('MT5 not connected. Make sure mt5_bridge.py is running.');
        return;
      }
      setMt5Connected(true);
      setMt5Error('');
      const [accountRes, openRes, summaryRes, accountsRes] = await Promise.all([
        fetch(`${MT5_URL}/account`),
        fetch(`${MT5_URL}/open-trades`),
        fetch(`${MT5_URL}/summary`),
        fetch(`${MT5_URL}/accounts`),
      ]);
      setMt5Account(await accountRes.json());
      setMt5OpenTrades(await openRes.json());
      setMt5Summary(await summaryRes.json());
      const accs = await accountsRes.json();
      setMt5Accounts(accs);
      const active = accs.findIndex(a => a.active);
      if (active >= 0) setMt5ActiveAccount(active);
    } catch (e) {
      setMt5Connected(false);
      setMt5Error('Cannot reach MT5 bridge. Make sure mt5_bridge.py is running.');
    }
  }, []);

  const fetchMt5History = useCallback(async (days) => {
    try {
      const res = await fetch(`${MT5_URL}/history/${days}`);
      setMt5History(await res.json());
    } catch (e) {}
  }, []);

  const switchMt5Account = useCallback(async (index) => {
    setMt5Switching(true);
    try {
      const res = await fetch(`${MT5_URL}/switch-account/${index}`, { method: 'POST' });
      const result = await res.json();
      if (result.success) {
        setMt5ActiveAccount(index);
        await fetchMt5Data();
        await fetchMt5History(mt5HistoryDays);
      } else {
        setMt5Error(result.error || 'Failed to switch account');
      }
    } catch (e) {
      setMt5Error('Failed to switch account');
    }
    setMt5Switching(false);
  }, [fetchMt5Data, fetchMt5History, mt5HistoryDays]);

  useEffect(() => {
    if (currentPage === 'mt5') {
      setMt5Loading(true);
      fetchMt5Data().finally(() => setMt5Loading(false));
      fetchMt5History(mt5HistoryDays);
      mt5RefreshRef.current = setInterval(fetchMt5Data, 5000);
    } else {
      clearInterval(mt5RefreshRef.current);
    }
    return () => clearInterval(mt5RefreshRef.current);
  }, [currentPage, fetchMt5Data, mt5HistoryDays]);

  // ===================== DATA LOADING =====================
  // Load trades on mount: prefer Electron file system, fallback to localStorage
  useEffect(() => {
    const loadData = async () => {
      try {
        if (window.electron?.loadTrades) {
          const fileTrades = await window.electron.loadTrades();
          if (fileTrades && fileTrades.length > 0) {
            setTrades(fileTrades);
            setDataLoaded(true);
            return;
          }
        }
        // Fallback to localStorage
        const saved = localStorage.getItem('trades');
        if (saved) {
          setTrades(JSON.parse(saved));
        }
      } catch (err) {
        console.error('Error loading trades:', err);
        const saved = localStorage.getItem('trades');
        if (saved) setTrades(JSON.parse(saved));
      }
      setDataLoaded(true);
    };
    loadData();
  }, []);

  // ===================== AUTO-SAVE FUNCTIONALITY =====================
  const saveTrades = useCallback(async (tradesToSave) => {
    setSaveStatus('saving');

    // Always save to localStorage as backup
    localStorage.setItem('trades', JSON.stringify(tradesToSave));

    // Save to Electron file system if available
    if (window.electron?.saveTrades) {
      try {
        const result = await window.electron.saveTrades(tradesToSave);
        if (result?.success) {
          setSaveStatus('saved');
        } else {
          setSaveStatus('error');
        }
      } catch (err) {
        console.error('Electron save error:', err);
        setSaveStatus('error');
      }
    } else {
      setSaveStatus('saved');
    }

    setShowSaveIndicator(true);
    if (saveTimeoutRef.current) clearTimeout(saveTimeoutRef.current);
    saveTimeoutRef.current = setTimeout(() => {
      setShowSaveIndicator(false);
    }, 2500);
  }, []);

  // Auto-save on trades change (debounced 2 seconds), skip on initial load
  useEffect(() => {
    if (!dataLoaded) return;
    if (autoSaveTimeoutRef.current) clearTimeout(autoSaveTimeoutRef.current);
    autoSaveTimeoutRef.current = setTimeout(() => {
      saveTrades(trades);
    }, 2000);
    return () => {
      if (autoSaveTimeoutRef.current) clearTimeout(autoSaveTimeoutRef.current);
    };
  }, [trades, saveTrades, dataLoaded]);

  // ===================== FORM HANDLERS =====================
  const handleFormChange = (e) => {
    const { name, value, checked, type } = e.target;
    if (name === 'confluences') {
      setFormData(prev => ({
        ...prev,
        confluences: checked
          ? [...prev.confluences, value]
          : prev.confluences.filter(c => c !== value)
      }));
    } else {
      setFormData(prev => ({
        ...prev,
        [name]: type === 'checkbox' ? checked : value
      }));
    }
  };

  const calculateMetrics = () => {
    if (formData.entryPrice && formData.exitPrice) {
      const entry = parseFloat(formData.entryPrice);
      const exit = parseFloat(formData.exitPrice);
      const pips = Math.abs(exit - entry) * 10000;
      setFormData(prev => ({ ...prev, pipsGained: pips.toFixed(1) }));
    }
  };

  const handleSaveTrade = () => {
    if (!formData.pair || !formData.entryPrice || !formData.exitPrice) {
      alert('Please fill in required fields: Pair, Entry Price, Exit Price');
      return;
    }
    if (editingId) {
      setTrades(prev => prev.map(t => t.id === editingId ? { ...formData, id: editingId } : t));
      setEditingId(null);
    } else {
      const newTrade = { ...formData, id: Date.now(), createdAt: new Date().toISOString() };
      setTrades(prev => [newTrade, ...prev]);
    }
    setFormData({ ...emptyForm, date: new Date().toISOString().split('T')[0] });
    setCurrentPage('trade-log');
  };

  const handleCancelEdit = () => {
    setEditingId(null);
    setFormData({ ...emptyForm, date: new Date().toISOString().split('T')[0] });
  };

  // ===================== ANALYTICS FUNCTIONS =====================
  const calculateStats = (tradeList) => {
    const list = tradeList || trades;
    if (list.length === 0) {
      return {
        totalTrades: 0, wins: 0, losses: 0, breakevens: 0,
        winRate: 0, totalProfit: 0, avgWin: 0, avgLoss: 0,
        profitFactor: 0, expectancy: 0, bestTrade: null,
        worstTrade: null, currentStreak: 0, bestStreak: 0,
        avgRR: 0, totalPips: 0,
      };
    }

    let wins = 0, losses = 0, breakevens = 0;
    let totalProfit = 0, totalWin = 0, totalLoss = 0;
    let bestTrade = null, worstTrade = null;
    let currentStreak = 0, bestStreak = 0;
    let totalPips = 0;

    list.forEach((trade) => {
      const profit = parseFloat(trade.riskAmount) || 0;
      const pips = parseFloat(trade.pipsGained) || 0;
      totalProfit += profit;
      totalPips += pips;

      if (trade.outcome === 'Win') {
        wins++;
        totalWin += Math.abs(profit);
        currentStreak++;
      } else if (trade.outcome === 'Breakeven') {
        breakevens++;
        currentStreak = 0;
      } else {
        losses++;
        totalLoss += Math.abs(profit);
        currentStreak = 0;
      }

      if (bestTrade === null || profit > (parseFloat(bestTrade.riskAmount) || 0)) bestTrade = trade;
      if (worstTrade === null || profit < (parseFloat(worstTrade.riskAmount) || 0)) worstTrade = trade;
      if (currentStreak > bestStreak) bestStreak = currentStreak;
    });

    const winRate = list.length > 0 ? ((wins / list.length) * 100).toFixed(2) : 0;
    const avgWin = wins > 0 ? (totalWin / wins).toFixed(2) : 0;
    const avgLoss = losses > 0 ? (totalLoss / losses).toFixed(2) : 0;
    const profitFactor = totalLoss > 0 ? (totalWin / totalLoss).toFixed(2) : totalWin > 0 ? '∞' : 0;
    const expectancy = list.length > 0 ? (totalProfit / list.length).toFixed(2) : 0;
    const avgRR = avgLoss > 0 ? (avgWin / avgLoss).toFixed(2) : 0;

    return {
      totalTrades: list.length, wins, losses, breakevens,
      winRate, totalProfit: totalProfit.toFixed(2), avgWin, avgLoss,
      profitFactor, expectancy, bestTrade, worstTrade,
      currentStreak, bestStreak, avgRR,
      totalPips: totalPips.toFixed(1),
    };
  };

  const getPairPerformance = () => {
    const pairStats = {};
    trades.forEach(trade => {
      if (!pairStats[trade.pair]) pairStats[trade.pair] = { wins: 0, losses: 0, profit: 0, trades: 0 };
      pairStats[trade.pair].trades++;
      if (trade.outcome === 'Win') pairStats[trade.pair].wins++;
      else if (trade.outcome === 'Loss') pairStats[trade.pair].losses++;
      pairStats[trade.pair].profit += parseFloat(trade.riskAmount) || 0;
    });
    return Object.entries(pairStats)
      .map(([pair, s]) => ({ pair, ...s, winRate: s.trades > 0 ? ((s.wins / s.trades) * 100).toFixed(1) : 0 }))
      .sort((a, b) => b.profit - a.profit);
  };

  const getSetupPerformance = () => {
    const setupStats = {};
    trades.forEach(trade => {
      if (!setupStats[trade.setup]) setupStats[trade.setup] = { wins: 0, losses: 0, profit: 0, trades: 0 };
      setupStats[trade.setup].trades++;
      if (trade.outcome === 'Win') setupStats[trade.setup].wins++;
      else if (trade.outcome === 'Loss') setupStats[trade.setup].losses++;
      setupStats[trade.setup].profit += parseFloat(trade.riskAmount) || 0;
    });
    return Object.entries(setupStats)
      .map(([setup, s]) => ({ setup, ...s, winRate: s.trades > 0 ? ((s.wins / s.trades) * 100).toFixed(1) : 0 }))
      .sort((a, b) => b.profit - a.profit);
  };

  const getEmotionPerformance = () => {
    const emotionStats = {};
    trades.forEach(trade => {
      if (!emotionStats[trade.emotion]) emotionStats[trade.emotion] = { wins: 0, losses: 0, profit: 0, trades: 0 };
      emotionStats[trade.emotion].trades++;
      if (trade.outcome === 'Win') emotionStats[trade.emotion].wins++;
      else if (trade.outcome === 'Loss') emotionStats[trade.emotion].losses++;
      emotionStats[trade.emotion].profit += parseFloat(trade.riskAmount) || 0;
    });
    return Object.entries(emotionStats)
      .map(([emotion, s]) => ({ emotion, ...s, winRate: s.trades > 0 ? ((s.wins / s.trades) * 100).toFixed(1) : 0 }))
      .sort((a, b) => b.winRate - a.winRate);
  };

  const getSessionPerformance = () => {
    const sessionStats = {};
    trades.forEach(trade => {
      if (!sessionStats[trade.session]) sessionStats[trade.session] = { wins: 0, losses: 0, profit: 0, trades: 0 };
      sessionStats[trade.session].trades++;
      if (trade.outcome === 'Win') sessionStats[trade.session].wins++;
      else if (trade.outcome === 'Loss') sessionStats[trade.session].losses++;
      sessionStats[trade.session].profit += parseFloat(trade.riskAmount) || 0;
    });
    return Object.entries(sessionStats)
      .map(([session, s]) => ({ session, ...s, winRate: s.trades > 0 ? ((s.wins / s.trades) * 100).toFixed(1) : 0 }))
      .sort((a, b) => b.profit - a.profit);
  };

  const getGradeDistribution = () => {
    const gradeStats = {};
    GRADES.forEach(g => { gradeStats[g] = 0; });
    trades.forEach(t => { if (gradeStats[t.grade] !== undefined) gradeStats[t.grade]++; });
    return gradeStats;
  };

  const getFilteredTradesForSummary = () => {
    const now = new Date();
    if (summaryPeriod === 'week') {
      const weekAgo = new Date(now);
      weekAgo.setDate(weekAgo.getDate() - 7);
      return trades.filter(t => new Date(t.date) >= weekAgo);
    } else if (summaryPeriod === 'month') {
      const monthAgo = new Date(now);
      monthAgo.setMonth(monthAgo.getMonth() - 1);
      return trades.filter(t => new Date(t.date) >= monthAgo);
    }
    return trades;
  };

  const getMonthlyData = () => {
    const monthly = {};
    trades.forEach(trade => {
      const month = trade.date?.substring(0, 7) || 'Unknown';
      if (!monthly[month]) monthly[month] = { trades: 0, wins: 0, profit: 0 };
      monthly[month].trades++;
      if (trade.outcome === 'Win') monthly[month].wins++;
      monthly[month].profit += parseFloat(trade.riskAmount) || 0;
    });
    return Object.entries(monthly)
      .sort(([a], [b]) => a.localeCompare(b))
      .slice(-6)
      .map(([month, data]) => ({
        month,
        ...data,
        winRate: data.trades > 0 ? ((data.wins / data.trades) * 100).toFixed(1) : 0
      }));
  };

  const stats = calculateStats();
  const filteredTrades = trades.filter(t => {
    const matchesPair = filterPair === 'All Pairs' || t.pair === filterPair;
    const matchesSearch = t.pair.toLowerCase().includes(searchTerm.toLowerCase()) ||
      (t.setup || '').toLowerCase().includes(searchTerm.toLowerCase()) ||
      (t.date || '').includes(searchTerm);
    return matchesPair && matchesSearch;
  });

  // ===================== RENDER: DASHBOARD =====================
  const renderDashboard = () => (
    <div className="dashboard">
      <div className="header-stats">
        <h1>📊 Trading Dashboard</h1>
        <div className="stat-grid">
          {[
            { label: 'Total Trades', value: stats.totalTrades, color: C.accent },
            { label: 'Wins', value: stats.wins, color: C.green },
            { label: 'Losses', value: stats.losses, color: C.red },
            { label: 'Win Rate %', value: `${stats.winRate}%`, color: C.blue },
            { label: 'Total P&L', value: `${stats.totalProfit >= 0 ? '+' : ''}${stats.totalProfit}`, color: parseFloat(stats.totalProfit) >= 0 ? C.green : C.red },
            { label: 'Profit Factor', value: stats.profitFactor, color: C.purple },
            { label: 'Expectancy', value: `$${stats.expectancy}`, color: C.accentLight },
            { label: 'Best Streak', value: stats.bestStreak, color: C.green },
          ].map((s, i) => (
            <div key={i} className="stat-box" style={{ borderColor: s.color }}>
              <div className="stat-label">{s.label}</div>
              <div className="stat-value" style={{ color: s.color }}>{s.value}</div>
            </div>
          ))}
        </div>
      </div>

      <div className="dashboard-grid">
        <div className="dashboard-section">
          <h2>📈 Top Performing Pairs</h2>
          <div className="table-container">
            <table>
              <thead>
                <tr style={{ borderBottom: `1px solid ${C.border}` }}>
                  <th>Pair</th><th>Trades</th><th>Win Rate</th><th>Profit</th>
                </tr>
              </thead>
              <tbody>
                {getPairPerformance().slice(0, 5).map((p, idx) => (
                  <tr key={idx} style={{ borderBottom: `1px solid ${C.border}` }}>
                    <td style={{ color: C.accent }}>{p.pair}</td>
                    <td>{p.trades}</td>
                    <td style={{ color: C.blue }}>{p.winRate}%</td>
                    <td style={{ color: p.profit >= 0 ? C.green : C.red }}>
                      {p.profit >= 0 ? '+' : ''}{p.profit.toFixed(2)}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

        <div className="dashboard-section">
          <h2>🎯 Setup Performance</h2>
          <div className="table-container">
            <table>
              <thead>
                <tr style={{ borderBottom: `1px solid ${C.border}` }}>
                  <th>Setup</th><th>Trades</th><th>Win Rate</th><th>Profit</th>
                </tr>
              </thead>
              <tbody>
                {getSetupPerformance().slice(0, 5).map((s, idx) => (
                  <tr key={idx} style={{ borderBottom: `1px solid ${C.border}` }}>
                    <td style={{ color: C.accent }}>{s.setup}</td>
                    <td>{s.trades}</td>
                    <td style={{ color: C.blue }}>{s.winRate}%</td>
                    <td style={{ color: s.profit >= 0 ? C.green : C.red }}>
                      {s.profit >= 0 ? '+' : ''}{s.profit.toFixed(2)}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      </div>

      {/* Recent 5 trades */}
      <div className="dashboard-section" style={{ marginTop: 30 }}>
        <h2>🕐 Recent Trades</h2>
        <div className="table-container">
          <table>
            <thead>
              <tr style={{ borderBottom: `1px solid ${C.border}` }}>
                <th>Date</th><th>Pair</th><th>Direction</th><th>Outcome</th><th>P&L</th><th>Grade</th>
              </tr>
            </thead>
            <tbody>
              {trades.slice(0, 5).map((trade) => (
                <tr key={trade.id} style={{ borderBottom: `1px solid ${C.border}` }}>
                  <td>{trade.date}</td>
                  <td style={{ color: C.accent }}>{trade.pair}</td>
                  <td style={{ color: trade.direction === 'Long' ? C.green : C.red }}>{trade.direction}</td>
                  <td style={{ color: trade.outcome === 'Win' ? C.green : trade.outcome === 'Loss' ? C.red : C.yellow, fontWeight: 'bold' }}>
                    {trade.outcome}
                  </td>
                  <td style={{ color: trade.outcome === 'Win' ? C.green : trade.outcome === 'Loss' ? C.red : C.yellow }}>
                    {trade.outcome === 'Win' ? '+' : ''}{trade.riskAmount || 0}
                  </td>
                  <td style={{ color: C.accent }}>{trade.grade}</td>
                </tr>
              ))}
            </tbody>
          </table>
          {trades.length === 0 && (
            <div style={{ textAlign: 'center', padding: 40, color: C.textSecondary }}>
              No trades yet. Click <strong style={{ color: C.accent }}>Log Trade</strong> to get started!
            </div>
          )}
        </div>
      </div>
    </div>
  );

  // ===================== RENDER: LOG TRADE =====================
  const renderLogTrade = () => (
    <div className="log-trade">
      <h1>{editingId ? '✏️ Edit Trade' : '📝 Log New Trade'}</h1>

      <div className="form-section">
        <h2>Trade Basics</h2>
        <div className="form-grid">
          {[
            { label: 'Date', name: 'date', type: 'date' },
            { label: 'Entry Time', name: 'entryTime', type: 'time' },
            { label: 'Exit Time', name: 'exitTime', type: 'time' },
          ].map(f => (
            <div key={f.name} className="form-group">
              <label>{f.label}</label>
              <input type={f.type} name={f.name} value={formData[f.name]} onChange={handleFormChange}
                style={{ background: C.bgInput, color: C.text, border: `1px solid ${C.border}` }} />
            </div>
          ))}
          <div className="form-group">
            <label>Pair</label>
            <select name="pair" value={formData.pair} onChange={handleFormChange}
              style={{ background: C.bgInput, color: C.text, border: `1px solid ${C.border}` }}>
              {PAIRS.map(p => <option key={p}>{p}</option>)}
            </select>
          </div>
          <div className="form-group">
            <label>Direction</label>
            <select name="direction" value={formData.direction} onChange={handleFormChange}
              style={{ background: C.bgInput, color: C.text, border: `1px solid ${C.border}` }}>
              <option>Long</option><option>Short</option>
            </select>
          </div>
          <div className="form-group">
            <label>Session</label>
            <select name="session" value={formData.session} onChange={handleFormChange}
              style={{ background: C.bgInput, color: C.text, border: `1px solid ${C.border}` }}>
              {SESSIONS.map(s => <option key={s}>{s}</option>)}
            </select>
          </div>
          <div className="form-group">
            <label>Timeframe</label>
            <select name="timeframe" value={formData.timeframe} onChange={handleFormChange}
              style={{ background: C.bgInput, color: C.text, border: `1px solid ${C.border}` }}>

    })();

    const emotionWinMap = {};

      if (!emotionWinMap[t.emotion]) emotionWinMap[t.emotion] = { wins: 0, total: 0 };
      emotionWinMap[t.emotion].total++;
      if (t.outcome === 'Win') emotionWinMap[t.emotion].wins++;
    });
    const bestEmotion = Object.entries(emotionWinMap)
      .filter(([, v]) => v.total >= 2)
      .sort(([, a], [, b]) => (b.wins / b.total) - (a.wins / a.total))[0];
    const worstEmotion = Object.entries(emotionWinMap)
      .filter(([, v]) => v.total >= 2)
      .sort(([, a], [, b]) => (a.wins / a.total) - (b.wins / b.total))[0];

    const setupWinMap = {};
    summaryTrades.forEach(t => {
      if (!setupWinMap[t.setup]) setupWinMap[t.setup] = { wins: 0, total: 0 };
      setupWinMap[t.setup].total++;
      if (t.outcome === 'Win') setupWinMap[t.setup].wins++;
    });
    const bestSetup = Object.entries(setupWinMap)
      .filter(([, v]) => v.total >= 2)
      .sort(([, a], [, b]) => (b.wins / b.total) - (a.wins / a.total))[0];

    const strengths = [];
    const improvements = [];

    else if (parseFloat(s.profitFactor) > 0) improvements.push(`Profit factor ${s.profitFactor} needs improvement (target 1.5+)`);

    if (parseFloat(s.avgRR) >= 2) strengths.push(`Average R:R of ${s.avgRR} shows disciplined reward-seeking`);
    else if (s.avgRR > 0) improvements.push(`Average R:R is ${s.avgRR} — try to target 2:1 or higher`);

    if (bestSetup) strengths.push(`Best setup is "${bestSetup[0]}" (${((bestSetup[1].wins / bestSetup[1].total) * 100).toFixed(0)}% win rate)`);

                Based on win rate, profit factor, R:R, trade quality & consistency

              {[
                { label: 'Trades', value: s.totalTrades, color: C.accent },
                { label: 'Wins', value: s.wins, color: C.green },

                { label: 'Breakevens', value: s.breakevens, color: C.yellow },
                { label: 'Win Rate', value: `${s.winRate}%`, color: parseFloat(s.winRate) >= 55 ? C.green : C.yellow },
                { label: 'Total P&L', value: `$${s.totalProfit}`, color: parseFloat(s.totalProfit) >= 0 ? C.green : C.red },
                { label: 'Profit Factor', value: s.profitFactor, color: parseFloat(s.profitFactor) >= 1.5 ? C.green : C.yellow },
                { label: 'Avg R:R', value: s.avgRR, color: parseFloat(s.avgRR) >= 2 ? C.green : C.yellow },
                { label: 'Expectancy', value: `$${s.expectancy}`, color: parseFloat(s.expectancy) >= 0 ? C.green : C.red },
                { label: 'Total Pips', value: s.totalPips, color: C.blue },
